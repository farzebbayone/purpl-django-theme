Welcome to Synth

This is a rich text editor powered by Quill, styled to match the Synth theme. It supports:

  • Bold, italic, and underline formatting
  • Ordered and unordered lists
  • Multiple heading levels
  • Code blocks and inline code

Integration
CDN Setup
<!-- In your template's extra_head block -->
<link href="https://cdn.jsdelivr.net/npm/quill@2/dist/quill.snow.css"
      rel="stylesheet">

<!-- In your template's extra_js block -->
<script src="https://cdn.jsdelivr.net/npm/quill@2/dist/quill.js">
</script>

<script>
const quill = new Quill('#editor', {
  theme: 'snow',
  placeholder: 'Write something...',
  modules: {
    toolbar: [
      [{ header: [2, 3, false] }],
      ['bold', 'italic', 'underline'],
      ['link', 'code-block'],
      [{ list: 'ordered' }, { list: 'bullet' }],
      ['clean']
    ]
  }
});
</script>
Django Form Sync
<!-- Hidden textarea for form submission -->
<form method="post">
  {% csrf_token %}
  <div id="editor"></div>
  <input type="hidden" name="content"
         id="content-field">
</form>

<script>
const quill = new Quill('#editor', {
  theme: 'snow'
});

// Sync editor content to hidden field
const form = document.querySelector('form');
form.addEventListener('submit', function() {
  document.getElementById('content-field')
    .value = quill.root.innerHTML;
});
</script>
